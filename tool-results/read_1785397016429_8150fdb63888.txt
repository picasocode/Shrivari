    23→    23→function FadeIn({ children, delay = 0, className = '' }: { children: React.ReactNode; delay?: number; className?: string }) {
    24→    24→  const ref = useRef(null)
    25→    25→  const isInView = useInView(ref, { once: true, margin: '-60px' })
    26→    26→  return (
    27→    27→    <motion.div
    28→    28→      ref={ref}
    29→    29→      initial={{ opacity: 0, y: 24 }}
    30→    30→      animate={isInView ? { opacity: 1, y: 0 } : {}}
    31→    31→      transition={{ duration: 0.5, delay, ease: 'easeOut' }}
    32→    32→      className={className}
    33→    33→    >
    34→    34→      {children}
    35→    35→    </motion.div>
    36→    36→  )
    37→    37→}

    39→    39→/* ─── Animated bar counter ─── */
    40→    40→function BarCounter({ value, suffix, label, maxVal, delay = 0, icon: Icon }: { value: number; suffix: string; label: string; maxVal: number; delay?: number; icon: React.ElementType }) {
    41→    41→  const [count, setCount] = useState(0)
    42→    42→  const [width, setWidth] = useState(0)
    43→    43→  const ref = useRef(null)
    44→    44→  const isInView = useInView(ref, { once: true })
    45→    45→
    46→    46→  useEffect(() => {
    47→    47→    if (!isInView) return
    48→    48→    const duration = 2000
    49→    49→    let startTime: number | null = null
    50→    50→    const step = (timestamp: number) => {
    51→    51→      if (!startTime) startTime = timestamp
    52→    52→      const progress = Math.min((timestamp - startTime) / duration, 1)
    53→    53→      const eased = 1 - Math.pow(1 - progress, 3)
    54→    54→      setCount(Math.floor(eased * value))
    55→    55→      setWidth(eased * (value / maxVal) * 100)
    56→    56→      if (progress < 1) requestAnimationFrame(step)
    57→    57→    }
    58→    58→    setTimeout(() => requestAnimationFrame(step), delay)
    59→    59→  }, [isInView, value, maxVal, delay])
    60→    60→
    61→    61→  return (
    62→    62→    <div ref={ref} className="group">
    63→    63→      <div className="flex items-center gap-3 mb-2">
    64→    64→        <div className="w-8 h-8 rounded-lg bg-[#E8751A]/15 flex items-center justify-center flex-shrink-0">
    65→    65→          <Icon className="w-4 h-4 text-[#E8751A]" />
    66→    66→        </div>
    67→    67→        <div className="flex-1 flex items-baseline justify-between">
    68→    68→          <span className="text-[#6B7280] text-sm font-medium">{label}</span>
    69→    69→          <span className="text-2xl md:text-3xl font-bold text-[#1A1A2E] tabular-nums">
    70→    70→            {count}{suffix}
    71→    71→          </span>
    72→    72→        </div>
    73→    73→      </div>
    74→    74→      <div className="h-2.5 bg-[#1F2937]/10 rounded-full overflow-hidden">
    75→    75→        <motion.div
    76→    76→          className="h-full rounded-full"
    77→    77→          style={{
    78→    78→            width: `${width}%`,
    79→    79→            background: 'linear-gradient(90deg, #E8751A 0%, #F59E0B 100%)',
    80→    80→          }}
    81→    81→        />
    82→    82→      </div>
    83→    83→    </div>
    84→    84→  )
    85→    85→}
