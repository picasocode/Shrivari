    19→/* ─── Fade-in helper ─── */
    20→function FadeIn({ children, delay = 0, className = '' }: { children: React.ReactNode; delay?: number; className?: string }) {
    21→  const ref = useRef(null)
    22→  const isInView = useInView(ref, { once: true, margin: '-60px' })
    23→  return (
    24→    <motion.div
    25→      ref={ref}
    26→      initial={{ opacity: 0, y: 24 }}
    27→      animate={isInView ? { opacity: 1, y: 0 } : {}}
    28→      transition={{ duration: 0.5, delay, ease: 'easeOut' }}
    29→      className={className}
    30→    >
    31→      {children}
    32→    </motion.div>
    33→  )
    34→}

    36→/* ─── Animated bar counter ─── */
    37→function BarCounter({ value, suffix, label, maxVal, delay = 0, icon: Icon }: { value: number; suffix: string; label: string; maxVal: number; delay?: number; icon: React.ElementType }) {
    38→  const [count, setCount] = useState(0)
    39→  const [width, setWidth] = useState(0)
    40→  const ref = useRef(null)
    41→  const isInView = useInView(ref, { once: true })
    42→
    43→  useEffect(() => {
    44→    if (!isInView) return
    45→    const duration = 2000
    46→    let startTime: number | null = null
    47→    const step = (timestamp: number) => {
    48→      if (!startTime) startTime = timestamp
    49→      const progress = Math.min((timestamp - startTime) / duration, 1)
    50→      const eased = 1 - Math.pow(1 - progress, 3)
    51→      setCount(Math.floor(eased * value))
    52→      setWidth(eased * (value / maxVal) * 100)
    53→      if (progress < 1) requestAnimationFrame(step)
    54→    }
    55→    setTimeout(() => requestAnimationFrame(step), delay)
    56→  }, [isInView, value, maxVal, delay])
    57→
    58→  return (
    59→    <div ref={ref} className="group">
    60→      <div className="flex items-center gap-3 mb-2">
    61→        <div className="w-8 h-8 rounded-lg bg-[#E8751A]/15 flex items-center justify-center flex-shrink-0">
    62→          <Icon className="w-4 h-4 text-[#E8751A]" />
    63→        </div>
    64→        <div className="flex-1 flex items-baseline justify-between">
    65→          <span className="text-white/60 text-sm font-medium">{label}</span>
    66→          <span className="text-2xl md:text-3xl font-bold text-white tabular-nums">
    67→            {count}{suffix}
    68→          </span>
    69→        </div>
    70→      </div>
    71→      <div className="h-2.5 bg-white/10 rounded-full overflow-hidden">
    72→        <motion.div
    73→          className="h-full rounded-full"
    74→          style={{
    75→            width: `${width}%`,
    76→            background: 'linear-gradient(90deg, #E8751A 0%, #F59E0B 100%)',
    77→          }}
    78→        />
    79→      </div>
    80→    </div>
    81→  )
    82→}
